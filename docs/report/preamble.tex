% --- the page ------------------------------------------------------------
% A4, two-sided margins in millimetres, set HERE and not in the front matter.
%
% The .qmd asks for `papersize: a4` and a geometry list, and Quarto passes
% neither to this template: the class options come out as `letterpaper` and
% the geometry package is never loaded at all. So the document was 612x792pt —
% US Letter — for its whole life while every margin in it was specified in
% millimetres and both READMEs called it A4. It was found by measuring the PDF
% rather than by reading the front matter, which is the only way this kind of
% thing is ever found, and it matters the moment anyone prints it.
%
% Loaded late, which is fine: geometry applies its page at \begin{document},
% so it wins over the class option regardless of order.
\usepackage[a4paper,top=26mm,bottom=26mm,inner=28mm,outer=24mm,twoside]{geometry}

% --- bleed and crop marks, for the press --------------------------------
% Off by default: the reading PDF is A4 trim, exactly. Turned on by
% docs/report/build.sh when BLEED=1, which writes a second file.
%
% Why it is needed at all: the sage band and the page canvas both run to the
% edge of the page. A printer handed trim-size artwork with nothing past the
% trim either asks for bleed or scales the page up to invent it, and scaling
% is how a 26mm margin quietly becomes 26.6mm. So the press file is drawn on
% larger paper with the page in the middle of it, the edge artwork carried
% 3mm past where the knife falls, and marks showing where that is.
%
%   trim   210 x 297      the A4 the reader gets
%   bleed  3mm            artwork past the knife on every side
%   marks  3mm gap, 4mm long, hairline
%   paper  230 x 317      trim + 2 x (bleed + gap + mark)
\newlength{\scbleed}   \setlength{\scbleed}{3mm}
\newlength{\scmarkgap} \setlength{\scmarkgap}{3mm}
\newlength{\scmarklen} \setlength{\scmarklen}{4mm}
\newif\ifbleed \bleedfalse
\IfFileExists{bleed.on}{\bleedtrue}{}
\ifbleed
  % geometry's own layout mechanism: the PAPER grows, the LAYOUT stays A4, and
  % every margin is still measured from the A4 the reader will hold.
  \geometry{papersize={230mm,317mm},
            layoutsize={210mm,297mm},
            layoutoffset={10mm,10mm}}
\fi

% --- Chinese.
% The report quotes Chinese statutory references and one string that must be
% reproduced verbatim (SAMR Order 37 Art. 12). Without a CJK font XeTeX drops
% those glyphs SILENTLY — the citations rendered as "2020 14 Art. 66(2)" with
% the characters simply gone, which in a legal-citation context is a defect
% rather than a cosmetic problem.
%
% Noto Serif SC, instanced to Regular and then subset to exactly the characters
% this document actually sets — 17 KB rather than 11.6 MB. fonts.py derives that
% set from the .qmd on every build, so adding a Chinese word to the prose cannot
% leave it uncovered.
\usepackage{xeCJK}
\setCJKmainfont{NotoSerifSC-report.otf}[Path=fonts/]
% The monospace slot has to be set too. Inside a verbatim environment xeCJK
% routes its character classes to the CJK MONO font; with none declared it fell
% through to the main CJK face, and every middle dot separating the file names
% in the build-tree listings was dropped without an error. Same subset, so the
% document still ships one Chinese face.
\setCJKmonofont{NotoSerifSC-report.otf}[Path=fonts/]
\xeCJKsetup{CJKmath=false}
% CJK/Latin glue. An earlier attempt set this to 0.08em to tighten bracket
% pairs such as 〔2020〕, which instead swallowed the word space in
% "文物before", "号Art." and "外汇under". 0.2em is the conventional value and
% keeps those readable; the slight looseness inside brackets is the lesser
% of the two problems.
\xeCJKsetup{xCJKecglue={\hskip 0.26em plus 0.06em minus 0.03em}}
% Keep Latin text in the Latin face; only CJK codepoints switch.
\xeCJKDeclareSubCJKBlock{CJK}{"3000 -> "303F, "3400 -> "4DBF, "4E00 -> "9FFF, "FF00 -> "FFEF}
% U+00B7 also sits in xeCJK's default CJK punctuation class, so a middle dot in
% running prose was taken from the Chinese face rather than the Latin one.
% Declare it an ordinary character; fonts.py keeps a copy in the subset as well,
% for the verbatim case above.
\xeCJKDeclareCharClass{Default}{"00B7}
